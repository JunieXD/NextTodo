-- holds them to. A new migration's part is added by `NEXTTODO_SCHEMA_ADD=1 cargo test -p nexttodo-server --lib migrations_keep`.

table dependencies:
    CREATE TABLE dependencies (
        before_id TEXT NOT NULL REFERENCES tasks (id) ON DELETE CASCADE,
        after_id TEXT NOT NULL REFERENCES tasks (id) ON DELETE CASCADE,
        owner_id TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        created_at INTEGER NOT NULL,
        PRIMARY KEY (before_id, after_id),
        CHECK (before_id <> after_id)
    ) STRICT
table folders:
    CREATE TABLE folders (
        id TEXT PRIMARY KEY,
        owner_id TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        parent_id TEXT REFERENCES folders (id) ON DELETE CASCADE,
        name TEXT NOT NULL,
        sort_key TEXT NOT NULL,
        created_at INTEGER NOT NULL,
        updated_at INTEGER NOT NULL
    ) STRICT
table invites:
    CREATE TABLE invites (
        code TEXT PRIMARY KEY,
        created_by TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        used_by TEXT REFERENCES users (id) ON DELETE SET NULL,
        created_at INTEGER NOT NULL,
        used_at INTEGER
    ) STRICT
table mutations:
    CREATE TABLE mutations (
        seq INTEGER PRIMARY KEY AUTOINCREMENT,
        id TEXT NOT NULL UNIQUE,
        owner_id TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        token_id TEXT NOT NULL,
        actor TEXT,
        -- all three cleared once past the retention period; the row stays so the id is still recognised
        ops TEXT,
        effects TEXT,
        -- what the tool that made the change answered, given again when the same call is retried
        reply TEXT,
        created_at INTEGER NOT NULL
    ) STRICT
table sqlite_sequence:
    CREATE TABLE sqlite_sequence(name,seq)
table tasks:
    CREATE TABLE tasks (
        id TEXT PRIMARY KEY,
        owner_id TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        number INTEGER NOT NULL,
        folder_id TEXT REFERENCES folders (id) ON DELETE CASCADE,
        title TEXT NOT NULL,
        description TEXT NOT NULL DEFAULT '',
        done_at INTEGER,
        due_date TEXT,
        due_time TEXT CHECK (due_time IS NULL OR due_date IS NOT NULL),
        remind_before INTEGER CHECK (remind_before IS NULL OR due_time IS NOT NULL),
        sort_key TEXT NOT NULL,
        created_at INTEGER NOT NULL,
        updated_at INTEGER NOT NULL,
        created_via TEXT,
        updated_via TEXT,
        UNIQUE (owner_id, number)
    ) STRICT
table tokens:
    CREATE TABLE tokens (
        id TEXT PRIMARY KEY,
        user_id TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        kind TEXT NOT NULL CHECK (kind IN ('session', 'api')),
        name TEXT NOT NULL,
        scope TEXT NOT NULL CHECK (scope IN ('read', 'write')),
        secret_hash BLOB NOT NULL UNIQUE,
        created_at INTEGER NOT NULL,
        last_used_at INTEGER
    ) STRICT
table users:
    CREATE TABLE users (
        id TEXT PRIMARY KEY,
        username TEXT NOT NULL UNIQUE COLLATE NOCASE,
        display_name TEXT NOT NULL,
        password_hash TEXT NOT NULL,
        is_admin INTEGER NOT NULL DEFAULT 0 CHECK (is_admin IN (0, 1)),
        next_task_number INTEGER NOT NULL DEFAULT 1,
        pruned_seq INTEGER NOT NULL DEFAULT 0,
        created_at INTEGER NOT NULL
    ) STRICT

    CREATE TABLE files (
        id TEXT PRIMARY KEY,
        owner_id TEXT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
        type TEXT NOT NULL,
        size INTEGER NOT NULL CHECK (size BETWEEN 1 AND 20971520),
        created_at INTEGER NOT NULL,
        unused_at INTEGER,
        -- Syncing through a drive (docs/local.md#附件): cleared, kept as a record so that its id is never
        -- used again; and uploaded on this device, not attached in a commit yet, this device's alone.
        purged INTEGER NOT NULL DEFAULT 0 CHECK (purged IN (0, 1)),
        staged INTEGER NOT NULL DEFAULT 0 CHECK (staged IN (0, 1)),
        -- Only a file no one uses is cleared, and then it is everyone's.
        CHECK (NOT purged OR (unused_at IS NOT NULL AND NOT staged))
    ) STRICT
